if(NOT TARGET crypto)
    include(${CMAKE_SOURCE_DIR}/cmake/LibreSSL.cmake)
endif()

add_library(libSceLibreSsl SHARED EXCLUDE_FROM_ALL Export.cpp)
target_include_directories(libSceLibreSsl PRIVATE ${LIBS_INCLUDE_DIR})
target_link_libraries(libSceLibreSsl PRIVATE crypto)
set_target_properties(libSceLibreSsl PROPERTIES VISIBILITY_INLINES_HIDDEN ON)
if(MINGW OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(libSceLibreSsl PRIVATE LINKER:--exclude-libs,ALL)
endif()
configure_windows_unwind(libSceLibreSsl)

if(BUILD_TESTING)
    add_test_executable(guest_libressl_tests ${LIBS_INCLUDE_DIR}/tests/GuestLibreSsl.cpp)
    target_include_directories(guest_libressl_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(guest_libressl_tests PRIVATE libSceLibreSsl libc)
    configure_windows_unwind(guest_libressl_tests)
    add_test(NAME guest_libressl COMMAND guest_libressl_tests)
    if(WIN32)
        set_tests_properties(guest_libressl PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceLibreSsl>")
    endif()
endif()
